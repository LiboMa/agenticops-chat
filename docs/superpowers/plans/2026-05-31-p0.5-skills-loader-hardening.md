```python
def build_available_skills_xml(skills: list[SkillMetadata]) -> str:
    """Generate <available_skills> XML block for agent system prompts.

    Descriptions and names are XML-escaped to prevent special characters
    (<, >, &, ") from breaking the system prompt. Full descriptions are
    available via activate_skill().
    """
    if not skills:
        return ""

    lines = ["<available_skills>"]
    for s in skills:
        short_desc = _xml_escape(s.description.split(".")[0][:80])
        safe_name = _xml_escape(s.name, {'"': "&quot;"})
        tag = "[DRAFT] " if s.is_draft else ""
        lines.append(f'  <skill name="{safe_name}">{tag}{short_desc}</skill>')
    lines.append("</available_skills>")
    return "\n".join(lines)
```

```python
def _fix_yaml_colons(yaml_str: str) -> str:
    """Quote unquoted values containing colons (mirrors Strands AgentSkills fallback).

    Handles authoring mistakes like:
        description: Use when: the user asks about PDFs
    which YAML treats as a nested mapping. Wraps such values in double quotes.
    """
    fixed_lines: list[str] = []
    for line in yaml_str.splitlines():
        m = re.match(r"^(\s*\w[\w-]*):\s+(.+)$", line)
        if m:
            key, value = m.group(1), m.group(2)
            if ":" in value and not (value.startswith('"') or value.startswith("'")):
                escaped = value.replace('"', '\\"')
                line = f'{key}: "{escaped}"'
        fixed_lines.append(line)
    return "\n".join(fixed_lines)
```

Replace `parse_frontmatter` body:

```python
def parse_frontmatter(content: str) -> tuple[dict, str]:
    """Parse YAML frontmatter from SKILL.md content.

    Falls back to colon-quoting repair if the initial yaml.safe_load fails —
    handles authoring mistakes like ``description: Use when: foo``.
    """
    match = _FRONTMATTER_RE.match(content)
    if not match:
        return {}, content

    raw = match.group(1)
    try:
        fm = yaml.safe_load(raw) or {}
    except yaml.YAMLError as e:
        logger.warning("YAML parse failed (%s); retrying with colon-quoting fallback", e)
        try:
            fm = yaml.safe_load(_fix_yaml_colons(raw)) or {}
        except yaml.YAMLError as e2:
            logger.warning("YAML fallback also failed: %s", e2)
            return {}, content

    if not isinstance(fm, dict):
        return {}, content

    body = content[match.end():]
    return fm, body
```

```python
def _validate_skill_name(name: str, skill_dir: Path) -> bool:
    """Validate skill name against the agentskills.io spec (lenient).

    Logs WARNING and returns False on any rule violation; the caller skips
    the skill. Never raises — keeps existing scan robustness.
    """
    if not name:
        logger.warning("Skill at %s has empty name — skipping", skill_dir)
        return False
    if len(name) > _MAX_SKILL_NAME_LEN:
        logger.warning("Skill name '%s' exceeds %d chars — skipping", name, _MAX_SKILL_NAME_LEN)
        return False
    if not _SKILL_NAME_RE.match(name):
        logger.warning("Skill name '%s' is not valid kebab-case — skipping (dir=%s)", name, skill_dir.name)
        return False
    if "--" in name:
        logger.warning("Skill name '%s' contains consecutive hyphens — skipping", name)
        return False
    if skill_dir.name != name:
        logger.warning("Skill name '%s' does not match parent directory '%s' — skipping", name, skill_dir.name)
        return False
    return True
```

```python
def list_skill_resources(skill_name: str) -> list[str]:
    """List relative paths under scripts/, references/, assets/.

    Mirrors Strands AgentSkills' resource listing so an agent activating a
    skill sees one menu of supporting files. Sorted; capped at
    ``_MAX_RESOURCE_FILES`` with a truncation marker on overflow.
    """
    s = _get_skill_by_name(skill_name)
    if s is None:
        return []
    files: list[str] = []
    for dir_name in _RESOURCE_DIRS:
        resource_dir = s.path / dir_name
        if not resource_dir.is_dir():
            continue
        for fp in sorted(resource_dir.rglob("*")):
            if not fp.is_file():
                continue
            files.append(fp.relative_to(s.path).as_posix())
            if len(files) >= _MAX_RESOURCE_FILES:
                files.append(f"... (truncated at {_MAX_RESOURCE_FILES} files)")
                return files
    return files
```

## Risks & Mitigations

| Risk | Mitigation |
|------|------------|
| Strict name validation rejects an existing skill | Lenient mode — logs WARNING, skips skill (never raises). Task 3 Step 5 explicitly verifies all 14 prod skills still load. If any rejected, fix the skill's frontmatter/directory — never relax the rule. |
| Concurrent pytest contention with main worktree | Task 6 Step 1 explicitly warns and instructs to check `ps` first. |
| YAML colon-fallback over-quotes legitimate dict values | Regex `^(\s*\w[\w-]*):\s+(.+)$` only matches scalar lines (key + value with text after). Lines starting nested mappings (`key:` alone) are untouched. Test `test_truly_broken_yaml_still_returns_empty` covers irrecoverable cases. |
| XML escape changes assertion text in unrelated tests | Run full `tests/test_skills_loader_coverage.py` after every task (Steps 5/6 in each task). Existing tests assert on plain ASCII descriptions — no entity-ref usage; should pass unchanged. |
| Description widening 80 → 200 increases prompt tokens | Per cycle③ spec §5: "先测量再决定". Net delta for 14 skills ≈ 14 × 120 chars ≈ 1.7 KB ≈ 425 tokens — well under the 2000-token threshold cycle③ already accepts. |
